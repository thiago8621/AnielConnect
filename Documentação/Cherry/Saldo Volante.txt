SELECT B.CODEMPRESA, D.EMPRESA, B.CODFILIAL, D.RAZAO_SOCIAL,
    A.CODCT, A.PROJETO, A.EQUIPE, A.CODMAT, SUM(A.QTDE) QTDE
FROM (
    SELECT A.CODCT, A.PROJETO, A.EQUIPE, A.CODMAT,
        (SUM(A.SAIDA) - SUM(A.ENTRADA)) AS QTDE
    FROM TB_ESTOQUE A
    LEFT JOIN TB_MATERIAL B ON (A.CODMAT = B.CODMAT)
    INNER JOIN TB_EQUIPE C ON (A.EQUIPE = C.EQUIPE AND C.ENCERRAMENTO IS NULL)
    WHERE A.ID_INV_EQUIPE IS NULL
    GROUP BY A.CODCT, A.PROJETO, A.EQUIPE, A.CODMAT
    HAVING (SUM(A.ENTRADA) - SUM(A.SAIDA)) <> 0

    UNION

    SELECT A.CODCT, A.PROJETO, A.EQUIPE, A.CODMAT,
        (SUM(A.QUANT_REMOV) - SUM(A.QUANTIDADE)) AS QTDE
    FROM TB_MATERIAL_TAREFA A
    LEFT JOIN TB_MATERIAL B ON (A.CODMAT = B.CODMAT)
    INNER JOIN TB_EQUIPE C ON (A.EQUIPE = C.EQUIPE AND C.ENCERRAMENTO IS NULL)
    WHERE A.ID_INV_EQUIPE IS NULL
    GROUP BY A.CODCT, A.PROJETO, A.EQUIPE, A.CODMAT
    HAVING (SUM(A.QUANT_REMOV) - SUM(A.QUANTIDADE)) <> 0) A
LEFT JOIN TB_CONTRATO B ON (A.CODCT = B.CODCT)
LEFT JOIN TB_FILIAL C ON (B.CODEMPRESA = C.CODEMPRESA AND B.CODFILIAL = C.CODFILIAL)
LEFT JOIN TB_EMPRESA D ON (B.CODEMPRESA = D.CODEMPRESA)
GROUP BY B.CODEMPRESA, D.EMPRESA, B.CODFILIAL, D.RAZAO_SOCIAL,
    A.CODCT, A.PROJETO, A.EQUIPE, A.CODMAT
