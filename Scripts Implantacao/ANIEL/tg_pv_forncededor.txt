CREATE OR ALTER trigger tg_pv_forncededor for tb_fornecedor
active after insert or update or delete position 0
AS
declare variable VCOD_FORNECEDOR VARCHAR(6);
declare variable VUSUARIO VARCHAR(10);
declare variable VACAO CHAR(1);
begin
  if (inserting) then
  BEGIN
    VACAO = 'I';
    VCOD_FORNECEDOR = NEW.COD_FORNECEDOR;
  END
  ELSE if (updating) then
  BEGIN
    VACAO = 'U';
    VCOD_FORNECEDOR = NEW.COD_FORNECEDOR;
  END
  ELSE
  BEGIN
    VACAO = 'D';
    VCOD_FORNECEDOR = OLD.COD_FORNECEDOR;
  END

  VUSUARIO = RDB$GET_CONTEXT('USER_SESSION', 'USUARIO');
  IF (VUSUARIO = '' OR VUSUARIO IS NULL) THEN
      VUSUARIO = 'DBA';

  insert into TB_PV_FORNECEDOR (ACAO, COD_FORNECEDOR, USUARIO, DIA, HORA)
  values (:VACAO, :VCOD_FORNECEDOR, :VUSUARIO, CURRENT_DATE, CURRENT_TIME);

end
